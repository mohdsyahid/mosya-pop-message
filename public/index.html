📦 Installation

// Option 1: CDN (Paling Mudah!)
<script src="https://cdn.jsdelivr.net/npm/mosya-pop-message@1.0.0/dist/index.umd.js"></script>

// Option 2: NPM
npm install mosya-pop-message
// Contoh Penggunaan:

// Simple Alert
MosyaPopMessage.alert({
  title: 'Hello World!',
  text: 'This is beautiful!',
  icon: 'success'
});

// Confirm Dialog
const result = await MosyaPopMessage.confirm({
  title: 'Are you sure?',
  text: 'Cannot undo this action!'
});

// Toast Notification  
MosyaPopMessage.toast.success('Operation completed!');

// Input Prompt
const value = await MosyaPopMessage.prompt({
  title: 'Enter your name',
  inputPlaceholder: 'John Doe...'
});

// Progress Indicator
const loadingId = MosyaPopMessage.progress({
  title: 'Processing...',
  text: 'Please wait...';
});
setTimeout(() => {
  MosyaPopMessage.hideProgress(loadingId);
}, 2000);
