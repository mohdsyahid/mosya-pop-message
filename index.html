💻 Quick Start

Basic Examples
// Simple Alert
MosyaPopMessage.alert({
  title: 'Hello World!',
  text: 'This is beautiful!',
  icon: 'success'
});

// Confirm Dialog
const result = await MosyaPopMessage.confirm({
  title: 'Are you sure?',
  text: 'Cannot undo this action!'
});

// Toast Notification  
MosyaPopMessage.toast.success('Operation completed!');

// Input Prompt
const value = await MosyaPopMessage.prompt({
  title: 'Enter your name',
  inputPlaceholder: 'John Doe...'
});

// Progress Indicator
const loadingId = MosyaPopMessage.progress({
  title: 'Processing...',
  text: 'Please wait...'
});
setTimeout(() => {
  MosyaPopMessage.hideProgress(loadingId);
}, 2000);
